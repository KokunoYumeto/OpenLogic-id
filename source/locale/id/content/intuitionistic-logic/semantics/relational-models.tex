\olsection{\usetoken{P}{relational model}}

Untuk memberikan semantik yang tepat bagi logika proposisional
intuisionistik, kita harus mendefinisikan apa yang terhitung sebagai model
untuk mengevaluasi !!{formula}s. Berdasarkan
definisi semacam itu, kita juga dapat mendefinisikan gagasan semantik seperti
kesahihan dan perikutan. Salah satu semantik tersebut diberikan oleh
!!{relational model}s.

\begin{defn}
  !!^a{relational model} bagi logika proposisional intuisionistik ialah suatu
  tripel $\mModel{M} = \tuple{W, R, V}$, dengan
  \begin{enumerate}
  \item $W$ suatu himpunan tak kosong,
  \item $R$ suatu urutan parsial (yakni, relasi biner yang refleksif,
    antisimetris, dan transitif) pada~$W$, dan
  \item $V$ suatu fungsi yang menetapkan kepada setiap !!{propositional
    variable}~$p$ suatu himpunan bagian dari~$W$, sedemikian sehingga
  \item $V$ monoton terhadap $R$; yakni, jika $w \in V(p)$ dan $Rww'$, maka
    $w' \in V(p)$.
  \end{enumerate}
\end{defn}

\begin{defn}\ollabel{defn:true-at-w}
  Kita mendefinisikan gagasan bahwa $!A$ \emph{benar pada $w$ dalam
    $\mModel{M}$}, $\mSat{M}{!A}[w]$, secara induktif sebagai berikut:
  \begin{enumerate}
  \item \indcase{!A}{p}{$\mSat{M}{\indfrm}[w]$ jika dan hanya jika $w \in V(p)$.}
  \item \indcase{!A}{\lfalse}{$\mSat{M}{\indfrm}[w]$ tidak berlaku}.
  \item \indcase{!A}{\lnot !B}{$\mSat{M}{\indfrm}[w]$ jika dan hanya jika
    tidak ada $w'$ sedemikian sehingga $Rww'$ dan $\mSat{M}{!B}[w']$}.
  \item \indcase{!A}{!B \land !C}{$\mSat{M}{\indfrm}[w]$ jika dan hanya jika
    $\mSat{M}{!B}[w]$ dan $\mSat{M}{!C}[w]$}.
  \item \indcase{!A}{!B \lor !C}{$\mSat{M}{\indfrm}[w]$ jika dan hanya jika
    $\mSat{M}{!B}[w]$ atau $\mSat{M}{!C}[w]$ (atau keduanya)}.
  \item \indcase{!A}{!B \lif !C}{$\mSat{M}{\indfrm}[w]$ jika dan hanya jika,
    bagi setiap $w'$ sedemikian sehingga $Rww'$, $\mSat{M}{!B}[w']$ tidak
    berlaku atau $\mSat{M}{!C}[w']$ berlaku} (atau keduanya).
  \end{enumerate}
  Kita menulis $\mSat/{M}{!A}[w]$ jika $\mSat{M}{!A}[w]$ tidak berlaku. Jika
  $\Gamma$ merupakan himpunan !!{formula}s, $\mSat{M}{\Gamma}[w]$ berarti
  $\mSat{M}{!B}[w]$ bagi semua $!B \in \Gamma$.
\end{defn}

\begin{prob}
  Tunjukkan bahwa menurut \olref[int][sem][rel]{defn:true-at-w},
  $\mSat{M}{\lnot !A}[w]$ jika dan hanya jika $\mSat{M}{!A \lif
  \lfalse}[w]$.
\end{prob}

\begin{prop}\ollabel{prop:true-monotonic}
  Kebenaran pada dunia bersifat monoton terhadap~$R$; yakni, jika
  $\mSat{M}{!A}[w]$ dan $Rww'$, maka $\mSat{M}{!A}[w']$.
\end{prop}

\begin{proof}
  Latihan.
\end{proof}

\begin{prob}
  Buktikan \olref[int][sem][rel]{prop:true-monotonic}.
\end{prob}
